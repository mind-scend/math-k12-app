```sql
-- ============================================
-- 读写分离：读从库专门处理复杂查询
-- ============================================

-- 写入主库
INSERT INTO user_behavior_logs (user_id, action_type, resource_id, duration_ms)

-- 复杂统计查询走从库
SELECT 
  grade,
  COUNT(*) as total_solves,
  AVG(duration_ms) as avg_duration,
  SUM(CASE WHEN result = 1 THEN 1 ELSE 0 END) / COUNT(*) as success_rate
FROM user_behavior_logs

  AND created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY)
GROUP BY grade;
```

### 3.1 索引设计策略

```javascript
// ============================================
// 题目集合：全文检索 + 复合索引
// ============================================

db.questions.createIndex(
  { "content": "text", "answer": "text" },
  { name: "idx_fulltext_search", weights: { content: 10, answer: 5 } }
);

// 复合索引：按年级+知识点+难度查询
db.questions.createIndex(
  { "grade": 1, "knowledge_points": 1, "difficulty": 1, "updated_at": -1 },
  { name: "idx_grade_knowledge_difficulty" }
);

// 热门题目索引：按热度排序
db.questions.createIndex(
  { "solve_count": -1, "difficulty": 1 },
  { name: "idx_hot_questions" }
);

// 相似题检索索引
db.questions.createIndex(
  { "embedding": "2dsphere" },  // 向量索引
  { name: "idx_semantic_search" }
);

// TTL索引：自动清理过期缓存
db.ai_cache.createIndex(
  { "created_at": 1 },
  { expireAfterSeconds: 86400 * 7, name: "idx_ttl_7days" }
);

// ============================================
// 错题本集合：用户高频访问优化
// ============================================

// 用户错题列表（按时间倒序）
db.wrong_questions.createIndex(
  { "user_id": 1, "created_at": -1 },
  { name: "idx_user_wrong_timeline" }
);

// 用户按知识点统计
db.wrong_questions.createIndex(
  { "user_id": 1, "knowledge_points": 1 },
  { name: "idx_user_knowledge", 
    // 部分文档聚合
    partialFilterExpression: { "is_mastered": { "$exists": false } }
  }
);

// 模糊搜索：支持题目内容搜索
db.wrong_questions.createIndex(
  { "content": "text", "knowledge_points": 1 },
  { name: "idx_user_wrong_search" }
);

// ============================================
// 生成的试卷集合：高效检索
// ============================================

db.generated_papers.createIndex(
  { "user_id": 1, "status": 1, "created_at": -1 },
  { name: "idx_user_paper_list" }
);

// AI生成题目去重索引
db.generated_papers.createIndex(
  { "source_question_id": 1, "user_id": 1 },
  { name: "idx_no_duplicate_generation" }
);
```

```
│                    请求链路                              │
└─────────────────────────────────────────────────────────┘
                          │
                          ▼
┌─────────────────────────────────────────────────────────┐
│  L1: 本地缓存（进程内）                                   │
│  - 用户会话：TTL 5分钟                                    │

│  - 命中率目标：40%                                       │
└─────────────────────────────────────────────────────────┘
                          │ 未命中
                          ▼
┌─────────────────────────────────────────────────────────┐
│  L2: Redis集群                                          │
│  - Session Token：TTL 24小时                             │
│  - 热门题目缓存：TTL 1天                                  │
│  - AI生成结果：TTL 7天                                   │
│  - 限流计数：TTL 1分钟滑动窗口                            │
│  - 命中率目标：30%                                       │
└─────────────────────────────────────────────────────────┘
                          │ 未命中
                          ▼
┌─────────────────────────────────────────────────────────┐
│  L3: MySQL/MongoDB                                     │
│  - 主数据库查询                                          │
│  - 命中率目标：30%                                       │
└─────────────────────────────────────────────────────────┘
```

### 4.2 Redis数据结构设计

```typescript
// ============================================
// Redis Key设计规范
// 命名规则：{业务}:{实体}:{维度}:{ID}
// ============================================

// 用户会话（String - Hash兼容）
SET  session:user:{user_id}  {JSON.stringify(session_data)}
EXPIRE session:user:{user_id}  86400  // 24小时

// 用户Token（String）
SET  token:{token}  {user_id}
EXPIRE token:{token}  604800  // 7天

// 热点题目缓存（Hash - 结构化存储）
HSET  question:hot:daily  Q-001  '{"id":"Q-001","content":"...","solved_count":9999}'
HSET  question:hot:daily  Q-002  '{"id":"Q-002","content":"...","solved_count":8888}'
EXPIRE  question:hot:daily  3600  // 每小时刷新

// AI解题结果缓存（String - 防重复请求）
SET  cache:solve:{md5(question)}  {JSON.stringify(result)}
EXPIRE  cache:solve:{md5(question)}  604800  // 7天

// 用户每日使用计数（String - 限流）
INCR  limit:daily:solve:{user_id}:{YYYYMMDD}
EXPIRE  limit:daily:solve:{user_id}:{YYYYMMDD}  86400

// 知识点掌握度（ZSet - 有序集合）
ZADD  mastery:user:{user_id}:{knowledge_id}  85  "2026-04-28"
ZREVRANGE  mastery:user:{user_id}:{knowledge_id}  0  0  // 获取最新掌握度

// 错题本列表（ZSet - 按时间排序）
ZADD  wrongbook:user:{user_id}  {timestamp}  {wrong_question_id}
ZREVRANGE  wrongbook:user:{user_id}  0  49  // 获取最近50条

// 分布式锁（String - 防并发）
SET  lock:generate:paper:{user_id}  {request_id}  NX  EX  300
// 解锁
DEL  lock:generate:paper:{user_id}

// 实时排行（ZSet - 点赞/收藏排行）
ZINCRBY  ranking:like:daily  1  "Q-001"
ZREVRANGE  ranking:like:daily  0  9  // 获取日榜Top10

// BloomFilter（防缓存穿透）
SETBIT  bloom:question:exists  {hash(question)}  1
EXISTS  bloom:question:exists  {hash(question)}  // 检查题目是否可能存在
```

### 4.3 缓存策略配置

```typescript
// ============================================
// 缓存策略配置文件
// ============================================

interface CacheConfig {
  // 缓存层级
  levels: [
    { name: 'local', ttl: 300, size: 1000 },    // 本地缓存5分钟
    { name: 'redis', ttl: 3600, size: 'unlimited' }  // Redis缓存1小时
  ];
  
  // 各业务缓存配置
  rules: [
    {
      pattern: '/api/user/profile',
      cache: true,
      ttl: 1800,  // 30分钟
      invalidate: ['update_profile']  // 更新时失效
    },
    {
      pattern: '/api/question/solve',
      cache: false,  // 解题结果不缓存（个性化）
      cacheHash: true,  // 但防重复计算
      ttl: 604800  // 7天
    },
    {
      pattern: '/api/paper/generate',
      cache: false,  // 每次生成不同
      lock: true,    // 加分布式锁
      ttl: 300  // 5分钟防重复
    },
    {
      pattern: '/api/wrongbook/list',
      cache: true,
      ttl: 300,
      invalidate: ['add_wrong', 'delete_wrong', 'update_mastery']
    }
  ];
}
```

```sql
  avg_timer_wait / 1000000000000 as avg_seconds,
  sum_timer_wait / 1000000000000 as total_seconds,
```

```javascript
// 自定义指标
const errorRate = new Rate('errors');

export const options = {
  stages: [
    { duration: '2m', target: 100 },   // 预热：2分钟到100用户
    { duration: '5m', target: 500 },  // 压测：5分钟到500用户

    http_req_duration: ['p95<500', 'p99<1000'],  // 95%请求<500ms
    errors: ['rate<0.05'],  // 错误率<5%
  },
};

const BASE_URL = __ENV.BASE_URL || 'http://localhost:3000';

export default function () {
  const userId = Math.floor(Math.random() * 100000);
  
  // 场景1：拍照解题（高频）

    `${BASE_URL}/api/solve`,

    'solve status 200': (r) => r.status === 200,

    'paper status 200': (r) => r.status === 200,
    'paper response time < 10s': (r) => r.timings.duration < 10000,
```
