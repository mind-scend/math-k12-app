## 二、核心数据模型

### 2.1 Question（题目主表）

```typescript
// TypeScript 类型定义
interface Question {
  // ===== 基础信息 =====
  _id: ObjectId;
  question_no: string;           // 题目编号 Q-2026-000001
  
  // ===== 内容字段 =====
  content: string;               // 题目内容（纯文本，含LaTeX公式）
  content_html: string;          // 题目HTML（带渲染公式）
  images: string[];              // 题目图片列表（如果有图）
  
  // ===== 答案与解析 =====
  answer: string;                // 标准答案
  answer_type: 'single' | 'multiple' | 'subjective';  // 答案类型
  solutions: Solution[];         // 多种解法
  
  // ===== 分类信息 =====
  subject: Subject;              // 学科
  grade: string;                 // 年级：初一、初二、初三、高一...
  chapter: string;               // 章节
  knowledge_points: string[];    // 知识点列表（支持多知识点）
  
  // ===== 难度与属性 =====
  difficulty: 1 | 2 | 3 | 4 | 5; // 难度等级
  question_type: QuestionType;   // 题型
  type_tags: string[];           // 类型标签：选择题、计算题、证明题...
  
  // ===== 来源追溯 =====
  source: QuestionSource;        // 来源
  source_id?: string;            // 来源ID（如果是AI生成，对应Prompt）
  source_question_id?: ObjectId; // 原始题目ID（用于举一反三溯源）
  
  // ===== 关联数据 =====
  similar_questions: ObjectId[]; // 相似题ID列表
  parent_question_id?: ObjectId; // 父题目ID（如果是变式题）
  variants: ObjectId[];          // 变式题列表
  
  // ===== 统计信息 =====
  stats: {
    usage_count: number;         // 使用次数
    correct_count: number;       // 正确次数
    wrong_count: number;         // 错误次数
    correct_rate: number;        // 历史正确率
    avg_score: number;           // 平均得分
    avg_time_seconds: number;    // 平均作答时间
  };
  
  // ===== 质量控制 =====
  status: 'pending' | 'approved' | 'rejected' | 'archived';
  quality_score?: number;        // 质量评分（AI评估）
  verified_by?: string;          // 审核人
  verified_at?: Date;
  
  // ===== 时间戳 =====
  created_at: Date;
  updated_at: Date;
  is_active: boolean;
}

interface Solution {
  method_name: string;           // 解法名称：如"配方法"、"换元法"
  steps: SolutionStep[];         // 步骤列表
  key_point: string;             // 关键点提示
  common_mistakes: string[];     // 常见错误提醒
}

interface SolutionStep {
  step_no: number;
  description: string;           // 步骤描述
  latex?: string;                // LaTeX公式
  image?: string;                // 步骤图示
  points: number;                // 该步得分
}

type Subject = 'math' | 'physics' | 'chemistry' | 'chinese' | 'english' | 'other';

type QuestionType = 
  | 'choice'           // 选择题
  | 'fill_blank'       // 填空题
  | 'judgment'         // 判断题
  | 'calculation'      // 计算题
  | 'proof'            // 证明题
  | 'application'      // 应用题
  | 'comprehensive'    // 综合题
  | 'short_answer';    // 简答题

type QuestionSource = 
  | 'question_bank'    // 题库采购
  | 'user_upload'      // 用户上传
  | 'ai_generated'     // AI生成
  | 'textbook_sync'    // 教材同步
  | 'exam_paper';      // 真题录入
```

### 2.2 WrongQuestion（用户错题）

```typescript
interface WrongQuestion {
  _id: ObjectId;
  
  // ===== 关联信息 =====
  user_id: ObjectId;             // 用户ID
  question_id: ObjectId;         // 原始题目ID
  
  // ===== 题目快照（核心！）=====
  question_snapshot: {
    content: string;
    answer: string;
    solutions: Solution[];
    knowledge_points: string[];
    difficulty: number;
    question_type: QuestionType;
    images?: string[];
  };
  
  // ===== 错题信息 =====
  wrong_answer?: string;         // 用户的错误答案
  
  // ===== 错因分析 =====
  error_type: ErrorType;
  error_reason: string;          // 详细错因描述
  ai_error_analysis?: string;    // AI分析的错因
  
  // ===== 掌握度追踪 =====
  mastery_level: number;         // 掌握度 0.0-1.0
  review_schedule: ReviewSchedule; // 复习计划
  
  // ===== 举一反三 =====
  similar_questions_generated: ObjectId[]; // AI生成的举一反三题
  practice_history: PracticeRecord[];
  
  // ===== 用户自定义 =====
  tags: string[];                // 用户标签
  notes?: string;                // 用户备注
  is_important: boolean;         // 是否标记重要
  
  // ===== 时间戳 =====
  created_at: Date;
  updated_at: Date;
  last_reviewed_at?: Date;
  next_review_at?: Date;
}

type ErrorType = 
  | 'calculation_error'      // 计算错误
  | 'concept_confusion'      // 概念混淆
  | 'wrong_method'           // 方法错误
  | 'careless_mistake'       // 粗心失误
  | 'incomplete_understanding' // 理解不完整
  | 'knowledge_gap'          // 知识盲区
  | 'reading_error';         // 审题错误

interface ReviewSchedule {
  // 艾宾浩斯遗忘曲线复习计划
  review_dates: Date[];
  current_index: number;         // 当前复习进度
  intervals: number[];           // 复习间隔天数：[1, 3, 7, 14, 30]
}

interface PracticeRecord {
  practice_date: Date;
  question_id: ObjectId;         // 练习的题目ID
  is_correct: boolean;
  time_spent: number;            // 花费时间（秒）
  user_answer?: string;
}
```

### 5.2 导入校验规则

```javascript
const importValidationRules = {
  required_fields: ['content', 'answer', 'subject', 'grade', 'knowledge_points'],
  
  field_validations: {
    content: {
      type: 'string',
      min_length: 5,
      max_length: 5000,
      pattern: /^[\s\S]*$/  // 允许LaTeX公式
    },
    answer: {
      type: 'string',
      min_length: 1,
      max_length: 2000
    },
    difficulty: {
      type: 'number',
      min: 1,
      max: 5,
      required: true
    },
    knowledge_points: {
      type: 'array',
      min_items: 1,
      max_items: 10,
      items_type: 'string'
    }
  },
  
  business_rules: [
    {
      rule: "选择题必须有选项",
      condition: (q) => q.question_type === 'choice' && !q.options
    },
    {
      rule: "答案类型与题型匹配",
      condition: (q) => {
        if (q.question_type === 'choice') {
          return ['A', 'B', 'C', 'D'].includes(q.answer)
        }
        return true
      }
    },
    {
      rule: "知识点必须在知识图谱中存在",
      condition: async (q) => {
        const existing = await db.knowledge_nodes.find({ 
          name: { $in: q.knowledge_points } 
        })
        return existing.length === q.knowledge_points.length
      }
    }
  ]
}
```

## 六、数据统计接口

### 6.1 常用统计查询

```javascript
// 1. 统计各难度题目数量
db.questions.aggregate([
  { $match: { subject: 'math', grade: '初三', status: 'approved' } },
  { $group: { _id: '$difficulty', count: { $sum: 1 } } },
  { $sort: { _id: 1 } }
])

// 2. 统计各知识点题目数量和平均正确率
db.questions.aggregate([
  { $unwind: '$knowledge_points' },
  { $match: { status: 'approved' } },
  { $group: {
      _id: '$knowledge_points',
      question_count: { $sum: 1 },
      avg_correct_rate: { $avg: '$stats.correct_rate' },
      avg_difficulty: { $avg: '$difficulty' }
    }
  },
  { $sort: { question_count: -1 } }
])

// 3. 统计用户错题分布
db.wrong_questions.aggregate([
  { $match: { user_id: ObjectId("用户ID") } },
  { $unwind: '$question_snapshot.knowledge_points' },
  { $group: {
      _id: '$question_snapshot.knowledge_points',
      count: { $sum: 1 },
      avg_mastery: { $avg: '$mastery_level' }
    }
  },
  { $sort: { count: -1 } }
])

// 4. 统计试卷生成情况
db.generated_papers.aggregate([
  { $match: { created_at: { $gte: ISODate("2026-04-01") } } },
  { $group: {
      _id: { $dateToString: { format: "%Y-%m-%d", date: "$created_at" } },
      paper_count: { $sum: 1 },
      avg_score: { $avg: '$grading_result.total_score' }
    }
  },
  { $sort: { _id: 1 } }
])
```
