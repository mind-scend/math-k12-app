```dockerfile
# backend/Dockerfile
FROM node:18-alpine AS builder

# 安装依赖

FROM node:18-alpine

WORKDIR /app

# 安全配置
RUN addgroup -g 1001 -S nodejs && adduser -S nodejs -u 1001

# 复制依赖
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/package*.json ./

# 复制源码
COPY --chown=nodejs:nodejs . .

# 切换用户
USER nodejs

# 健康检查
HEALTHCHECK --interval=30s --timeout=10s --start-period=5s --retries=3 \
  CMD wget --no-verbose --tries=1 --spider http://localhost:3000/health || exit 1

EXPOSE 3000

# 信号处理
CMD ["sh", "-c", "trap 'kill -TERM $PID' TERM INT; node src/index.js & PID=$!; wait"]

# 或使用pm2
# CMD ["npx", "pm2-runtime", "start", "ecosystem.config.js"]
```

### 2.2 Nginx配置

```nginx
# nginx.conf
worker_processes auto;
error_log /var/log/nginx/error.log warn;
pid /var/run/nginx.pid;

events {
    worker_connections 1024;
    use epoll;
    multi_accept on;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    # 日志格式
    log_format main '$remote_addr - $remote_user [$time_local] "$request" '
                    '$status $body_bytes_sent "$http_referer" '
                    '"$http_user_agent" "$http_x_forwarded_for" '
                    'rt=$request_time uct="$upstream_connect_time" '
                    'uht="$upstream_header_time" urt="$upstream_response_time"';

    access_log /var/log/nginx/access.log main;

    # 性能优化
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;

    # Gzip压缩
    gzip on;
    gzip_vary on;
    gzip_proxied any;
    gzip_comp_level 6;
    gzip_types text/plain text/css text/xml application/json application/javascript 
                application/xml application/xml+rss text/javascript application/x-javascript;

    # 上传大小限制
    client_max_body_size 10M;

    # IP限流
    limit_req_zone $binary_remote_addr zone=api:10m rate=100r/s;
    limit_conn_zone $binary_remote_addr zone=addr:10m;

    # IP白名单
    geo $whitelist {
        default 0;
        127.0.0.1 1;
        10.0.0.0/8 1;
    }

    upstream backend {
        least_conn;
        server api1:3001 weight=5;
        server api2:3002 weight=5;
        keepalive 32;
    }

    server {
        listen 80;
        server_name api.mathk12.com;

        # SSL配置
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256;
        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;
        ssl_session_cache shared:SSL:10m;
        ssl_session_timeout 1d;

        # 安全头
        add_header X-Frame-Options "SAMEORIGIN" always;
        add_header X-Content-Type-Options "nosniff" always;
        add_header X-XSS-Protection "1; mode=block" always;
        add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

        location / {
            root /usr/share/nginx/html;
            index index.html;
            try_files $uri $uri/ /index.html;
        }

        location /api/ {
            # 限流
            limit_req zone=api burst=200 nodelay;
            limit_conn addr 50;

            # 代理配置
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # 超时配置
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;

            # WebSocket支持
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
        }

        location /health {
            access_log off;
            return 200 "healthy\n";
            add_header Content-Type text/plain;
        }
    }
}
```

### 3.1 GitHub Actions工作流

```yaml
# .github/workflows/ci-cd.yml
name: MathK12 CI/CD Pipeline

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}

jobs:
  # ============================================
  # Job 1: 代码检查
  # ============================================
  lint:
    name: Code Lint
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '18'
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Run ESLint
        run: npm run lint

      - name: Run Prettier check
        run: npm run format:check

  # ============================================
  # Job 2: 单元测试
  # ============================================
  test:
    name: Unit Tests
    runs-on: ubuntu-latest
    needs: lint
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '18'
          cache: 'npm'

      - name: Install dependencies
        run: npm ci

      - name: Run unit tests
        run: npm run test:coverage

      - name: Upload coverage
        uses: codecov/codecov-action@v3
        with:
          files: ./coverage/lcov.info
          flags: unittests

  # ============================================
  # Job 3: 构建Docker镜像
  # ============================================
  build:
    name: Build Docker Image
    runs-on: ubuntu-latest
    needs: test
    if: github.event_name == 'push'
    outputs:
      image_tag: ${{ steps.meta.outputs.tags }}
      image_digest: ${{ steps.build.outputs.digest }}
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Setup Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Log in to Container Registry
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Extract metadata
        id: meta
        uses: docker/metadata-action@v5
        with:
          images: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}
          tags: |
            type=ref,event=branch
            type=sha,prefix={{branch}}-
            type=raw,value=latest,enable={{is_default_branch}}

      - name: Build and push
        uses: docker/build-push-action@v5
        with:
          context: ./backend
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          labels: ${{ steps.meta.outputs.labels }}
          cache-from: type=gha
          cache-to: type=gha,mode=max

  # ============================================
  # Job 4: 部署到测试环境
  # ============================================
  deploy-staging:
    name: Deploy to Staging
    runs-on: ubuntu-latest
    needs: build
    if: github.ref == 'refs/heads/develop'
    environment: staging
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Deploy to staging server
        uses: appleboy/ssh-action@master
        with:
          host: ${{ secrets.STAGING_HOST }}
          username: ${{ secrets.STAGING_USER }}
          key: ${{ secrets.STAGING_SSH_KEY }}
          script: |
            cd /opt/mathk12
            docker-compose pull api
            docker-compose up -d api
            docker image prune -f

  # ============================================
  # Job 5: 部署到生产环境
  # ============================================
  deploy-production:
    name: Deploy to Production
    runs-on: ubuntu-latest
    needs: build
    if: github.ref == 'refs/heads/main'
    environment: production
    steps:
      - name: Checkout code
```

### 3.2 部署脚本

```bash
#!/bin/bash
# deploy.sh - 生产环境部署脚本

set -e

ENV=${1:-production}
REGISTRY="ghcr.io/mathk12"
IMAGE_TAG=${2:-latest}

echo "========================================"
echo "开始部署 MathK12 ($ENV)"
echo "镜像: $REGISTRY/api:$IMAGE_TAG"
echo "========================================"

# 备份函数
backup_database() {
    echo "📦 备份数据库..."
    TIMESTAMP=$(date +%Y%m%d_%H%M%S)
    
    # MongoDB备份
    docker exec mathk12-mongo-1 mongodump \
      --out=/backup/mongo_$TIMESTAMP \
      --archive=/backup/mongo_$TIMESTAMP.archive
    
    # 上传到OSS
    # mc cp local/file oss/bucket/path/
    
    echo "✅ 备份完成: mongo_$TIMESTAMP"
}

# 回滚函数
rollback() {
    echo "⚠️ 回滚到上一版本..."
    docker-compose pull api
    docker-compose up -d api
    exit 1
}

# 部署函数
deploy() {
    echo "📥 拉取最新镜像..."
    docker pull $REGISTRY/api:$IMAGE_TAG
    
    echo "🚀 停止旧容器..."
    docker-compose stop api
    
    echo "🔄 启动新容器..."
    docker-compose up -d api
    
    echo "⏳ 等待服务启动..."
    sleep 10
    
    echo "🔍 健康检查..."
    for i in {1..30}; do
        if curl -sf http://localhost:3000/health > /dev/null; then
            echo "✅ 健康检查通过"
            return 0
        fi
        echo "尝试 $i/30..."
        sleep 2
    done
    
    echo "❌ 健康检查失败"
    return 1
}

# 主流程
main() {
    # 备份
    backup_database
    
    # 部署
    if ! deploy; then
        rollback
    fi
    
    # 清理旧镜像
    echo "🧹 清理旧镜像..."
    docker image prune -f
    
    echo "========================================"
    echo "✅ 部署完成！"
    echo "========================================"
}

main
```

## 五、安全加固

### 5.1 服务器安全加固

```bash
#!/bin/bash
# security-hardening.sh - 服务器安全加固脚本

set -e

echo "========================================"
echo "开始服务器安全加固"
echo "========================================"

# 1. 更新系统
apt-get update && apt-get upgrade -y

# 2. 配置防火墙
ufw default deny incoming
ufw default allow outgoing
ufw allow 22/tcp    # SSH
ufw allow 80/tcp    # HTTP
ufw allow 443/tcp   # HTTPS
ufw --force enable

# 3. SSH安全配置
sed -i 's/#PermitRootLogin yes/PermitRootLogin no/' /etc/ssh/sshd_config
sed -i 's/#PasswordAuthentication yes/PasswordAuthentication no/' /etc/ssh/sshd_config
sed -i 's/#MaxAuthTries 6/MaxAuthTries 3/' /etc/ssh/sshd_config
systemctl restart sshd

# 4. 安装fail2ban防暴力破解
apt-get install -y fail2ban
systemctl enable fail2ban
systemctl start fail2ban

# 5. 配置sysctl加固
cat >> /etc/sysctl.conf << EOF
# SYN洪水攻击防护
net.ipv4.tcp_syncookies = 1
net.ipv4.tcp_syn_retries = 2
net.ipv4.tcp_synack_retries = 2

# 禁止IP转发
net.ipv4.ip_forward = 0
net.ipv6.conf.all.forwarding = 0

# 禁止ICMP重定向
net.ipv4.conf.all.accept_redirects = 0
net.ipv4.conf.default.accept_redirects = 0
EOF

sysctl -p

# 6. 禁止ping
echo "net.ipv4.icmp_echo_ignore_all = 1" >> /etc/sysctl.conf
sysctl -p

# 7. 日志审计
cat >> /etc/rsyslog.conf << EOF
# Auth日志独立文件
auth.* /var/log/auth.log
authpriv.* /var/log/auth.log
EOF

systemctl restart rsyslog

# 8. 设置日志轮转
cat > /etc/logrotate.d/mathk12 << EOF
/var/log/mathk12/*.log {
    daily
    missingok
    rotate 30
    compress
    delaycompress
    notifempty
    create 0640 www-data adm
    sharedscripts
    postrotate
        systemctl reload rsyslog > /dev/null 2>&1 || true
    endscript
}
EOF

echo "========================================"
echo "✅ 安全加固完成"
echo "========================================"
```

```javascript
// src/middleware/security.js

const helmet = require('helmet');
```

### 6.2 恢复流程

```bash
#!/bin/bash
# restore.sh - 数据恢复脚本

set -e

BACKUP_TYPE=${1:-mongodb}  # mongodb | redis | files
BACKUP_FILE=${2:-latest}    # 指定文件或latest

echo "========================================"
echo "开始恢复: $BACKUP_TYPE"
echo "备份文件: $BACKUP_FILE"
echo "========================================"

# 确认操作
read -p "⚠️ 确认要恢复数据吗？这将覆盖当前数据！(y/n): " confirm
if [ "$confirm" != "y" ]; then
    echo "取消恢复操作"
    exit 0
fi

case $BACKUP_TYPE in
  mongodb)
    # 下载备份
    echo "📥 下载MongoDB备份..."
    mc cp oss/mathk12-backup/mongodb/$BACKUP_FILE /tmp/mongo_backup.archive.gz

    # 创建临时恢复数据库
    echo "🗄️ 创建临时恢复数据库..."
    mongorestore --uri="mongodb://localhost:27017/temp_restore" \
      --archive=/tmp/mongo_backup.archive.gz \
      --gzip \
      --drop

    # 验证数据
    echo "🔍 验证数据..."
    mongosh --quiet --eval "db.adminCommand('ping')" temp_restore

    # 执行切换（新旧数据库原子性切换）
    echo "⚡ 执行数据库切换..."
    mongosh --quiet --eval "
      use admin;
      db.adminCommand({
        renameCollection: 'temp_restore.questions',
        to: 'education_k12.questions'
      });
    "
    ;;

  redis)
    echo "📥 下载Redis备份..."
    mc cp oss/mathk12-backup/redis/$BACKUP_FILE /tmp/redis_backup.rdb

    echo "⏹️ 停止Redis服务..."
    docker-compose stop redis

    echo "📁 替换RDB文件..."
    mv /tmp/redis_backup.rdb /data/dump.rdb

    echo "▶️ 启动Redis服务..."
    docker-compose start redis
    ;;

  files)
    echo "📥 恢复文件..."
    mc mirror oss/mathk12-backup/files/$BACKUP_FILE /opt/mathk12/uploads/
    ;;
esac

echo "========================================"
echo "✅ 恢复完成！"
echo "========================================"

# 发送通知
curl -X POST "$SLACK_WEBHOOK" \
  -H 'Content-Type: application/json' \
  -d '{"text": "✅ MathK12数据恢复完成\n类型: '$BACKUP_TYPE'\n文件: '$BACKUP_FILE'"}'
```

### 6.3 RTO/RPO目标
